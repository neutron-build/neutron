# Files permitted to contain unsafe Rust. Every entry is reviewed; adding one
# means someone read the SAFETY invariant, not that the check was noisy.
#
# Checked by scripts/check-unsafe.sh, which greps for `unsafe fn`,
# `unsafe impl`, and `unsafe {`.

# SIMD intrinsics — target-feature-gated, checked at runtime before dispatch.

src/simd/avx2.rs
src/simd/avx512.rs
src/simd/mod.rs
src/vector/mod.rs

# Low-level storage: frame memory in the buffer pool, io_uring submission,
# and `fsync(2)` FFI for the weaker-than-F_FULLFSYNC sync mode.
src/storage/buffer.rs
src/storage/io_uring.rs
src/storage/wal.rs

# `libc::statvfs` FFI for the free-space watermark guard.
src/ops/disk.rs

# Misc reviewed call sites.
src/executor/session.rs
src/config/mod.rs
